dbo_destiny_talented_youngest_heir_selection_check = {
	trigger_if = {
		limit = {
			succession_law = heir_selection:dbo_destiny_talented_youngest_heir_selection
		}
		succession_law = heir_selection:dbo_destiny_talented_youngest_heir_selection
	}
	trigger_else = {
		has_ruler = yes
		has_heir = yes
	}
}

dbo_destiny_anatolian_beyliks_forced_annex_check = {
	has_variable = rise_of_the_ottomans_winner_variable
	any_country = {
		has_reform = government_reform:anatolian_beylik
		NOT = {
			tag = TUR
		}
	}
}

dbo_destiny_ottoman_empire_check = {
	trigger_if = {
		limit = {
			has_reform = government_reform:dbo_destiny_ottoman_empire
		}
		has_reform = government_reform:dbo_destiny_ottoman_empire
	}
	trigger_else = {
		country_rank = country_rank:rank_empire
		OR = {
			has_reform = government_reform:legacy_of_the_ottomans
			has_reform = government_reform:turkish_pashas
		}
	}
}
